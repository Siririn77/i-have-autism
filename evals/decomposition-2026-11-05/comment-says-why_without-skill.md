```python
def seconds_to_time(seconds: int) -> str:
    """
    Переводит количество секунд в строку вида "1ч 20м".
    Секунды, не образующие полную минуту, отбрасываются.
    """
    # Отрицательное количество секунд не имеет смысла в данном формате
    if seconds < 0:
        raise ValueError("Количество секунд не может быть отрицательным")

    # Считаем полное количество минут, отбрасывая оставшиеся секунды
    total_minutes = seconds // 60

    # Делим минуты на часы и оставшиеся минуты
    hours, minutes = divmod(total_minutes, 60)

    # Собираем части результата
    parts = []

    # Добавляем часы, если они есть
    if hours > 0:
        parts.append(f"{hours}ч")

    # Добавляем минуты, если они есть,
    # или если часов нет вообще — чтобы вернуть хотя бы "0м"
    if minutes > 0 or not parts:
        parts.append(f"{minutes}м")

    # Объединяем части через пробел
    return " ".join(parts)


# Примеры использования:
print(seconds_to_time(4800))   # 1ч 20м
print(seconds_to_time(3600))   # 1ч
print(seconds_to_time(90))     # 1м
print(seconds_to_time(30))     # 0м
```

Если нужно всегда показывать минуты, даже когда они равны нулю, можно заменить условие для минут на:

```python
if minutes > 0 or hours > 0:
    parts.append(f"{minutes}м")
```

Тогда `3600` секунд выведутся как `1ч 0м`.
